Tab:AddButton({
    Name = "Kill Selected Player",
    Callback = function()
        local selectedPlayer = getgenv().selectedPlayer  -- تأكد من أن اسم اللاعب الذي تم اختياره موجود هنا
        local player = game.Players.LocalPlayer  -- هذه هي شخصيتك

        if selectedPlayer then
            local targetPlayer = game.Players:FindFirstChild(selectedPlayer)  -- إيجاد اللاعب الذي تم اختياره

            if targetPlayer then
                -- نقل الأداة "Couch" من الـ Backpack إلى الشخصية في كل مرة يتم الضغط فيها على الزر
                local couchInBackpack = nil
                -- التحقق إذا كانت الأداة "Couch" موجودة في الـ Backpack
                for _, item in pairs(player.Backpack:GetChildren()) do
                    if item.Name == "Couch" then
                        couchInBackpack = item
                        break
                    end
                end

                -- إذا كانت الأداة "Couch" موجودة في الـ Backpack، نقوم بنقلها إلى الـ Character
                if couchInBackpack then
                    couchInBackpack.Parent = player.Character  -- نقل الأداة إلى الشخصية
                    print("تم نقل الأداة 'Couch' إلى الشخصية.")
                else
                    -- إذا كانت الأداة "Couch" غير موجودة في الـ Backpack، نقوم بتفعيلها
                    local args = {[1] = "PickingTools", [2] = "Couch"}
                    game:GetService("ReplicatedStorage").RE:FindFirstChild("1Too1l"):InvokeServer(unpack(args))
                    print("تم تفعيل الأداة: Couch")
                end

                -- الانتقال إلى اللاعب المحدد (الذي تم اختياره) ولكن مع تعديل الموقع ليكون تحت اللاعب قليلاً
                if targetPlayer.Character and targetPlayer.Character.PrimaryPart then
                    -- تعديل موقع CFrame بحيث تكون الشخصية الخاصة بك تحت اللاعب قليلاً
                    local targetPosition = targetPlayer.Character.PrimaryPart.Position
                    local newPosition = targetPosition - Vector3.new(0, 5, 0)  -- تقليل الإحداثيات في المحور Y ليكون تحت اللاعب قليلاً
                    game.Players.LocalPlayer.Character:SetPrimaryPartCFrame(CFrame.new(newPosition))
                    print("تم الانتقال إلى: " .. selectedPlayer)
                end

                -- دوران الشخصية بشكل متقلب وكامل حول جميع المحاور
                local startTime = tick()  -- وقت البداية
                local oscillationTime = 0  -- وقت التذبذب
                while tick() - startTime < .6 do  -- التكرار لمدة 2 ثانية
                    if player.Character and player.Character.PrimaryPart then
                        -- تدوير الشخصية بشكل سريع حول المحاور الثلاثة (X، Y، Z) ليكون مثل حركة "Fling"
                        player.Character:SetPrimaryPartCFrame(player.Character.PrimaryPart.CFrame * CFrame.Angles(
                            math.rad(math.random(500, 1000)),  -- تدوير عشوائي حول محور X
                            math.rad(math.random(500, 1000)),  -- تدوير عشوائي حول محور Y
                            math.rad(math.random(500, 1000))   -- تدوير عشوائي حول محور Z
                        ))

                        -- تحريك الشخصية للأعلى والأسفل (التذبذب)
                        local yOffset = math.sin(oscillationTime) * 5  -- استخدام sin للتذبذب
                        player.Character:SetPrimaryPartCFrame(player.Character.PrimaryPart.CFrame * CFrame.new(0, yOffset, 0))
                        oscillationTime = oscillationTime + 0.1  -- زيادة الزمن لجعل التذبذب مستمر
                    end
                    wait(0.001)  -- الانتظار لمدة 0.01 ثانية لضمان سلاسة الحركة
                end

                -- الانتظار لمدة 0.5 ثانية قبل التحرك إلى مكان بعيد
                wait(0.5)

                -- تحريك الشخصية إلى مكان بعيد جدًا خارج الماب
                local outsideMapPosition = Vector3.new(10000, 0, 10000)  -- تغيير القيم لهذه النقطة بعيدة
                game.Players.LocalPlayer.Character:SetPrimaryPartCFrame(CFrame.new(outsideMapPosition))
                print("تم نقل الشخصية إلى مكان بعيد.")

                -- الانتظار لمدة قصيرة قبل العودة
                wait(0.5)

                -- بعد العودة من المكان البعيد، نبحث عن الأداة "Couch" في الشخصية
                local couchInCharacter = nil
                if player.Character then
                    for _, item in pairs(player.Character:GetChildren()) do
                        if item.Name == "Couch" then
                            couchInCharacter = item
                            break
                        end
                    end
                end

                -- إذا كانت الأداة "Couch" موجودة في الشخصية، نقوم بإرجاعها إلى الـ Backpack
                if couchInCharacter then
                    couchInCharacter.Parent = player.Backpack  -- إرجاع الأداة إلى الـ Backpack
                    print("تم إعادة الأداة 'Couch' إلى الـ Backpack.")
                end

                -- العودة إلى المكان الأصلي
                if targetPlayer.Character and targetPlayer.Character.PrimaryPart then
                    local returnPosition = targetPlayer.Character.PrimaryPart.Position - Vector3.new(0, 5, 0)  -- العودة إلى مكان تحت اللاعب
                    game.Players.LocalPlayer.Character:SetPrimaryPartCFrame(CFrame.new(returnPosition))
                    print("تم العودة إلى المكان الأصلي.")
                end
            end
        end
    end    
})

local infiniteJumpEnabled = false

-- Conectar o evento de pulo somente uma vez
local infiniteJumpConnection

-- Função de callback para o botão de alternância de pulo infinito
local function onInfiniteJumpToggle(value)

infiniteJumpEnabled = value
    print("Infinite Jump Enabled:", infiniteJumpEnabled)
    
    -- Conectar o evento de pulo somente uma vez
    if not infiniteJumpConnection then
        infiniteJumpConnection = game:GetService("UserInputService").JumpRequest:Connect(function()
            if infiniteJumpEnabled then
                local player = game.Players.LocalPlayer
                local character = player.Character
                if character and character:FindFirstChildOfClass("Humanoid") then
                    character:FindFirstChildOfClass("Humanoid"):ChangeState("Jumping")
                end
            end
        end)
    end
end

-- Adiciona o botão de alternância "Infinitejump"
Tab:AddToggle({
    Name = "Infinite Jump",
    Default = false,
    Callback = onInfiniteJumpToggle
})

local runService = game:GetService("RunService")

-- دالة لتعديل CanCollide
local function setCharacterCanCollide(character, value)
    for _, part in pairs(character:GetChildren()) do
        if part:IsA("BasePart") then
            part.CanCollide = value
        end
    end
end

-- دالة لتفعيل / تعطيل الـ Noclip
local function onNoclipToggle(value)
    noclipEnabled = value
    print("Noclip Enabled:", noclipEnabled)
    
    local player = game.Players.LocalPlayer
    local character = player.Character

    if noclipEnabled then
        -- بدء حلقة لتحديث CanCollide باستمرار
        noclipLoop = runService.Stepped:Connect(function()
            if character then
                setCharacterCanCollide(character, false)
            end
        end)
    else
        -- تعطيل الحلقة واستعادة CanCollide
        if noclipLoop then
            noclipLoop:Disconnect()
        end
        if character then
            setCharacterCanCollide(character, true)
        end
    end
end
-- إضافة التبديل إلى الـ Tab
Tab:AddToggle({
    Name = "No Clip",
    Default = false,
    Callback = onNoclipToggle
})

local players = {} -- قائمة اللاعبين

-- ملء القائمة بأسماء اللاعبين في السيرفر
for _, player in pairs(game.Players:GetPlayers()) do
    table.insert(players, player.Name)
end

Tab:AddDropdown({
    Name = "Players",
    Default = "1", 
    Options = players, 
    Callback = function(Value)
        getgenv().selectedPlayer = Value -- تخزين اللاعب المحدد في متغير عام
    end    
})

Tab:AddButton({
    Name = "Copy Player Avatar",
    Callback = function()
        local selectedPlayerName = getgenv().selectedPlayer
        if not selectedPlayerName then
            warn("يرجى اختيار لاعب أولاً!")
            return
        end
        
        -- الحصول على اللاعب المستهدف
        local selectedPlayer = game.Players:FindFirstChild(selectedPlayerName)
        if not selectedPlayer or not selectedPlayer.Character then
            warn("لا يمكن العثور على شخصية هذا اللاعب!")
            return
        end

        local targetCharacter = selectedPlayer.Character
        local localPlayer = game.Players.LocalPlayer
        local localCharacter = localPlayer.Character
        
        if not localCharacter then
            warn("شخصيتك غير موجودة!")
            return
        end

        -- إزالة الملابس والإكسسوارات القديمة
        for _, item in pairs(localCharacter:GetChildren()) do
            if item:IsA("Shirt") or item:IsA("Pants") or item:IsA("Accessory") then
                item:Destroy()
            end
        end

        -- تطبيق الملابس والإكسسوارات الجديدة
        for _, item in pairs(targetCharacter:GetChildren()) do
            if item:IsA("Shirt") or item:IsA("Pants") or item:IsA("Accessory") then
                local clonedItem = item:Clone()
                clonedItem.Parent = localCharacter
            end
        end

        -- نسخ أجزاء الجسم (لون الجلد، مشهد الشخصية، إلخ)
        for _, part in pairs(targetCharacter:GetChildren()) do
            if part:IsA("CharacterMesh") or part:IsA("BodyColors") then
                local clonedPart = part:Clone()
                clonedPart.Parent = localCharacter
            end
        end

        -- نسخ الأنيميشن (إذا كان موجودًا)
        local targetAnimate = targetCharacter:FindFirstChild("Animate")
        if targetAnimate then
            local clonedAnimate = targetAnimate:Clone()
            local oldAnimate = localCharacter:FindFirstChild("Animate")
            if oldAnimate then
                oldAnimate:Destroy()
            end
            clonedAnimate.Parent = localCharacter
        end

        -- تحديث الشخصية لضمان تطبيق التغييرات
        local humanoid = localCharacter:FindFirstChildOfClass("Humanoid")
        if humanoid then
            humanoid:ApplyDescription(humanoid:GetAppliedDescription())
        end
    end
})

game.Players.PlayerAdded:Connect(function(player)
    table.insert(players, player.Name)
    Tab:RefreshDropdown("Players", players) -- تحديث القائمة تلقائيًا
end)

game.Players.PlayerRemoving:Connect(function(player)
    for i, name in ipairs(players) do
        if name == player.Name then
            table.remove(players, i)
            break
        end
    end
    Tab:RefreshDropdown("Players", players) -- تحديث القائمة بعد خروج لاعب
end)

Tab:AddButton({
    Name = "نسخ مظهر اللاعب بالكامل",
    Callback = function()
        local selectedPlayerName = getgenv().selectedPlayer
        if not selectedPlayerName then
            warn("يرجى اختيار لاعب أولاً!")
            return
        end
        
        -- الحصول على اللاعب المستهدف
        local selectedPlayer = game.Players:FindFirstChild(selectedPlayerName)
        if not selectedPlayer or not selectedPlayer.Character then
            warn("لا يمكن العثور على شخصية هذا اللاعب!")
            return
        end
        
        local targetCharacter = selectedPlayer.Character
        local humanoid = targetCharacter:FindFirstChildOfClass("Humanoid")
        
        if not humanoid then
            warn("لا يمكن العثور على Humanoid في الشخصية!")
            return
        end

        -- استخراج HumanoidDescription الخاص باللاعب المستهدف
        local description = humanoid:GetAppliedDescription()

        -- إرسال مظهر الجسم بناءً على الـ Catalog ID الخاصة بالأجزاء
        local argsBody = {
            [1] = "CharacterChange",
            [2] = {
                [1] = description.Torso,        -- جسم اللاعب (الجذع)
                [2] = description.RightArm,     -- الذراع اليمنى
                [3] = description.LeftArm,      -- الذراع اليسرى
                [4] = description.RightLeg,     -- الساق اليمنى
                [5] = description.LeftLeg,      -- الساق اليسرى
                [6] = description.Head          -- رأس اللاعب
            },
            [3] = "Roblox20"
        }
        -- إرسال بيانات الجسم
        game:GetService("ReplicatedStorage").RE:FindFirstChild("1Avata1rOrigina1l"):FireServer(unpack(argsBody))

        -- دالة لتغيير المظهر باستخدام الـ Catalog ID وطباعة الـ Catalog ID مع اسم العنصر
        local function changeWear(wearType, catalogId, itemName)
            print("تم تغيير " .. itemName .. " باستخدام Catalog ID: " .. tostring(catalogId))  -- طباعة اسم العنصر مع الـ Catalog ID
            local args = {
                [1] = "wear",    -- لتغيير الـ wear
                [2] = catalogId  -- Catalog ID الخاص بالملابس أو الإكسسوار
            }
            -- إرسال البيانات عبر FireServer
            local success, errorMsg = pcall(function()
                game:GetService("ReplicatedStorage").RE:FindFirstChild("1Updat1eAvata1r"):FireServer(unpack(args))
            end)
            
            -- التحقق من حالة النجاح أو الفشل
            if success then
                print(itemName .. " تم تغييره بنجاح!")
            else
                warn("حدث خطأ أثناء تغيير " .. itemName .. ": " .. errorMsg)
            end
        end

        -- تغيير القميص والبنطال بناءً على الـ Catalog ID في الوصف
        if description.Shirt and description.Shirt ~= "" then
            changeWear("wearShirt", description.Shirt, "القميص")  -- تغيير القميص باستخدام الـ Catalog ID الخاص بالقميص
        end
        if description.Pants and description.Pants ~= "" then
            changeWear("wearPants", description.Pants, "البنطال")  -- تغيير البنطال باستخدام الـ Catalog ID الخاص بالبناطيل
        end

        -- تغيير الشعر بناءً على الـ Catalog ID الخاص بالشعر
        if description.HairAccessory and description.HairAccessory ~= "" then
            changeWear("wearHair", description.HairAccessory, "الشعر")  -- تغيير الشعر باستخدام الـ Catalog ID الخاص بالشعر
        end

        -- تغيير القبعة بناءً على الـ Catalog ID الخاص بالقبعة
        if description.HatAccessory and description.HatAccessory ~= "" then
            changeWear("wearHat", description.HatAccessory, "القبعة")  -- تغيير القبعة باستخدام الـ Catalog ID الخاص بالقبعة
        end

        -- تغيير الوجه بناءً على الـ Catalog ID الخاص بالوجه
        if description.FaceAccessory and description.FaceAccessory ~= "" then
            changeWear("wearFace", description.FaceAccessory, "الوجه")  -- تغيير الوجه باستخدام الـ Catalog ID الخاص بالوجه
        end

        -- إضافة الأنيميشن (مثل حركة المشي) - يمكن تعديلها حسب الحاجة
        local argsAnimation = {
            [1] = "wearWalkStyle",  -- لتغيير حركة المشي
            [2] = description.WalkSpeed  -- استخدام سرعة المشي للاعب أو الأنيميشن
        }
        game:GetService("ReplicatedStorage").RE:FindFirstChild("1Updat1eAvata1r"):FireServer(unpack(argsAnimation))

        -- إضافة باقي الإكسسوارات مثل النظارات أو الإكسسوارات الأخرى
        local accessories = {}
        for _, accessory in pairs(targetCharacter:GetChildren()) do
            if accessory:IsA("Accessory") then
                table.insert(accessories, accessory)
            end
        end

        -- تغيير الإكسسوارات الأخرى بناءً على الـ Catalog ID
        for _, accessory in ipairs(accessories) do
            if accessory.AssetId then
                changeWear("wear", accessory.AssetId, "الإكسسوار: " .. accessory.Name)  -- تغيير الإكسسوار باستخدام الـ Catalog ID
            end
        end
    end
})
